-- Q08. Trọng tâm vĩ độ bản ghi Animalia theo lớp và thập kỷ
-- Latitude Sum sums latitude at fact-row grain. All accepted facts have coordinates.

WITH
MEMBER [Measures].[Latitude Centroid] AS
    IIF([Measures].[Record Count] > 0,
        [Measures].[Latitude Sum] / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.0000"

MEMBER [Measures].[Sampled Sites] AS
    COUNT(NONEMPTY(EXISTING [Location].[Coordinates].[Coordinates].MEMBERS,
                   {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

SELECT
    {[Measures].[Latitude Centroid], [Measures].[Record Count],
     [Measures].[Sampled Sites]} ON COLUMNS,
    NONEMPTY(
        EXCEPT([Taxon].[Class].[Class].MEMBERS,
               {[Taxon].[Class].&[__UNKNOWN__]})
        * [Date].[Decade].[Decade].MEMBERS,
        {[Measures].[Record Count]}) ON ROWS
FROM [Marine Range Shift]
WHERE ([Taxon].[Kingdom].&[Animalia]);
