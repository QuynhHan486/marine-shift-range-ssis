-- Q04. Ba lớp Animalia có nhiều loài nhất trong lát cắt đã chọn
-- Rank once across pooled decades/bands, then compare the same three classes.
-- Subselect restricts real members before TOPCOUNT; never sum slice-specific richness.

WITH
SET [Known Species] AS
    EXCEPT([Taxon].[Species].[Species].MEMBERS,
           {[Taxon].[Species].&[__UNKNOWN__]})

MEMBER [Measures].[Recorded Species] AS
    COUNT(NONEMPTY(EXISTING [Known Species], {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

SET [Known Classes] AS
    EXCEPT([Taxon].[Class].[Class].MEMBERS,
           {[Taxon].[Class].&[__UNKNOWN__]})

SET [Top Three Classes] AS
    TOPCOUNT(NONEMPTY([Known Classes], {[Measures].[Record Count]}),
             3, [Measures].[Recorded Species])

SELECT
    {[Measures].[Recorded Species], [Measures].[Record Count]} ON COLUMNS,
    NONEMPTY([Top Three Classes]
             * {[Date].[Decade].&[2000], [Date].[Decade].&[2010]}
             * {[Location].[Latitude Band].&[10], [Location].[Latitude Band].&[20]},
             {[Measures].[Record Count]}) ON ROWS
FROM (
    SELECT
        {[Date].[Decade].&[2000], [Date].[Decade].&[2010]}
        * {[Location].[Latitude Band].&[10], [Location].[Latitude Band].&[20]}
        * {[Depth Band].[Band].&[1], [Depth Band].[Band].&[2]} ON COLUMNS
    FROM [Marine Range Shift]
)
WHERE ([Taxon].[Kingdom].&[Animalia]);
