-- Adapted to deployed cube MARINE SHIFT RANGE DW / database MARINE_SHIFT_RANGE_SSAS.
-- Q02. Mùa theo lịch được ghi nhận nhiều nhất của mỗi giới
-- Winter=12/1/2; Spring=3/4/5; Summer=6/7/8; Autumn=9/10/11.
-- TOPCOUNT returns one winning season. A remaining tie is not returned.

WITH
MEMBER [Measures].[Record Count] AS
    [Measures].[Occurrence Count], FORMAT_STRING = "#,##0"

MEMBER [Measures].[Season Record Share] AS
    IIF(([Measures].[Record Count], [Dim Date].[Season].DEFAULTMEMBER) > 0,
        COALESCEEMPTY([Measures].[Record Count], 0) / ([Measures].[Record Count], [Dim Date].[Season].DEFAULTMEMBER), NULL),
    FORMAT_STRING = "0.00%"

SELECT
    {[Measures].[Record Count], [Measures].[Season Record Share]} ON COLUMNS,
    GENERATE(
        FILTER([Dim Taxon].[Kingdom].[Kingdom].MEMBERS,
           LEN(TRIM([Dim Taxon].[Kingdom].CURRENTMEMBER.PROPERTIES("KEY0"))) > 0),
        TOPCOUNT(
            NONEMPTY({[Dim Taxon].[Kingdom].CURRENTMEMBER}
                     * [Dim Date].[Season].[Season].MEMBERS,
                     {[Measures].[Record Count]}),
            1, [Measures].[Record Count])
    ) ON ROWS
FROM [MARINE SHIFT RANGE DW];
