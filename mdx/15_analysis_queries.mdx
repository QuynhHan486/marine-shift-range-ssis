-- Q01. Độ đầy đủ và cờ chất lượng theo thập kỷ
-- Taxonomy completeness is measured after Dim_Taxon standardization.
-- Flag Consistency means flag_on_land agrees with the ON_LAND token.
-- This checks warehouse data, not the original unfiltered OBIS export.

WITH
SET [Known Species] AS
    EXCEPT([Taxon].[Species].[Species].MEMBERS,
           {[Taxon].[Species].&[__UNKNOWN__]})

MEMBER [Measures].[Recorded Species] AS
    COUNT(NONEMPTY(EXISTING [Known Species], {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Sampled Sites] AS
    COUNT(NONEMPTY(EXISTING [Location].[Coordinates].[Coordinates].MEMBERS,
                   {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Depth Known Records] AS
    COALESCEEMPTY(SUM(
        {[Depth Band].[Band].&[1], [Depth Band].[Band].&[2],
         [Depth Band].[Band].&[3], [Depth Band].[Band].&[4]},
        [Measures].[Record Count]), 0),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Species Identified Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY(SUM([Known Species], [Measures].[Record Count]), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Complete Taxonomy Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY(([Measures].[Record Count], [Taxon].[Taxonomy Complete].&[1]), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Country Known Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY([Measures].[Record Count] - COALESCEEMPTY(([Measures].[Record Count], [Location].[Country].&[__UNKNOWN__]), 0), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Depth Known Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY([Measures].[Depth Known Records], 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[SST Known Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY([Measures].[Record Count] - COALESCEEMPTY(([Measures].[Record Count], [Location].[SST Group].&[-1]), 0), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[SSS Known Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY([Measures].[Record Count] - COALESCEEMPTY(([Measures].[Record Count], [Location].[SSS Group].&[-1]), 0), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Flagged Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY([Measures].[Record Count] - COALESCEEMPTY(([Measures].[Record Count], [Quality].[Flags].&[[]]]), 0), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[On Land Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY(([Measures].[Record Count], [Quality].[On Land].&[1]), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Flag Consistency Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY(([Measures].[Record Count], [Quality].[Flag Consistency].&[1]), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Depth Issue Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY(([Measures].[Record Count], [Quality].[Depth Issue].&[1]), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Accepted Name Issue Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY(([Measures].[Record Count], [Quality].[Accepted Name Issue].&[1]), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

SELECT
    {[Measures].[Record Count], [Measures].[Recorded Species], [Measures].[Sampled Sites],
     [Measures].[Species Identified Pct], [Measures].[Complete Taxonomy Pct],
     [Measures].[Country Known Pct], [Measures].[Depth Known Pct],
     [Measures].[SST Known Pct], [Measures].[SSS Known Pct],
     [Measures].[Flagged Pct], [Measures].[On Land Pct],
     [Measures].[Flag Consistency Pct], [Measures].[Depth Issue Pct],
     [Measures].[Accepted Name Issue Pct]} ON COLUMNS,
    NONEMPTY([Date].[Decade].[Decade].MEMBERS,
             {[Measures].[Record Count]}) ON ROWS
FROM [Marine Range Shift];


-- Q02. Mùa theo lịch được ghi nhận nhiều nhất của mỗi giới
-- Winter=12/1/2; Spring=3/4/5; Summer=6/7/8; Autumn=9/10/11.
-- TOPCOUNT returns one winning season. A remaining tie is not returned.

WITH
MEMBER [Measures].[Season Record Share] AS
    IIF(([Measures].[Record Count], [Date].[Season].DEFAULTMEMBER) > 0,
        COALESCEEMPTY([Measures].[Record Count], 0) / ([Measures].[Record Count], [Date].[Season].DEFAULTMEMBER), NULL),
    FORMAT_STRING = "0.00%"

SELECT
    {[Measures].[Record Count], [Measures].[Season Record Share]} ON COLUMNS,
    GENERATE(
        EXCEPT([Taxon].[Kingdom].[Kingdom].MEMBERS,
               {[Taxon].[Kingdom].&[__UNKNOWN__]}),
        TOPCOUNT(
            NONEMPTY({[Taxon].[Kingdom].CURRENTMEMBER}
                     * [Date].[Season].[Season].MEMBERS,
                     {[Measures].[Record Count]}),
            1, [Measures].[Record Count])
    ) ON ROWS
FROM [Marine Range Shift];

-- Q04. Ba lớp Animalia có nhiều loài nhất trong lát cắt đã chọn
-- Rank once across pooled decades/bands, then compare the same three classes.
-- Subselect restricts real members before TOPCOUNT; never sum slice-specific richness.

WITH
SET [Known Species] AS
    EXCEPT([Taxon].[Species].[Species].MEMBERS,
           {[Taxon].[Species].&[__UNKNOWN__]})

MEMBER [Measures].[Recorded Species] AS
    COUNT(NONEMPTY(EXISTING [Known Species], {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

SET [Known Classes] AS
    EXCEPT([Taxon].[Class].[Class].MEMBERS,
           {[Taxon].[Class].&[__UNKNOWN__]})

SET [Top Three Classes] AS
    TOPCOUNT(NONEMPTY([Known Classes], {[Measures].[Record Count]}),
             3, [Measures].[Recorded Species])

SELECT
    {[Measures].[Recorded Species], [Measures].[Record Count]} ON COLUMNS,
    NONEMPTY([Top Three Classes]
             * {[Date].[Decade].&[2000], [Date].[Decade].&[2010]}
             * {[Location].[Latitude Band].&[10], [Location].[Latitude Band].&[20]},
             {[Measures].[Record Count]}) ON ROWS
FROM (
    SELECT
        {[Date].[Decade].&[2000], [Date].[Decade].&[2010]}
        * {[Location].[Latitude Band].&[10], [Location].[Latitude Band].&[20]}
        * {[Depth Band].[Band].&[1], [Depth Band].[Band].&[2]} ON COLUMNS
    FROM [Marine Range Shift]
)
WHERE ([Taxon].[Kingdom].&[Animalia]);

-- Q07. Năm ô lưới có nhiều loài nhất mỗi thập kỷ
-- GENERATE emits decade+grid tuples, so each decade gets its own ranking.
-- Sampled Sites counts coordinate pairs, not location surrogate IDs.
-- TOPCOUNT excludes remaining ties; only cells with at least one identified species qualify.

WITH
SET [Known Species] AS
    EXCEPT([Taxon].[Species].[Species].MEMBERS,
           {[Taxon].[Species].&[__UNKNOWN__]})

MEMBER [Measures].[Recorded Species] AS
    COUNT(NONEMPTY(EXISTING [Known Species], {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Sampled Sites] AS
    COUNT(NONEMPTY(EXISTING [Location].[Coordinates].[Coordinates].MEMBERS,
                   {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

SELECT
    {[Measures].[Recorded Species], [Measures].[Record Count],
     [Measures].[Sampled Sites]} ON COLUMNS,
    GENERATE(
        [Date].[Decade].[Decade].MEMBERS,
        TOPCOUNT(
            FILTER(
                NONEMPTY({[Date].[Decade].CURRENTMEMBER}
                         * [Location].[Grid Cell].[Grid Cell].MEMBERS,
                         {[Measures].[Record Count]}),
                [Measures].[Recorded Species] > 0),
            5, [Measures].[Recorded Species])
    ) ON ROWS
FROM [Marine Range Shift];


-- Q08. Trọng tâm vĩ độ bản ghi Animalia theo lớp và thập kỷ
-- Latitude Sum sums latitude at fact-row grain. All accepted facts have coordinates.

WITH
MEMBER [Measures].[Latitude Centroid] AS
    IIF([Measures].[Record Count] > 0,
        [Measures].[Latitude Sum] / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.0000"

MEMBER [Measures].[Sampled Sites] AS
    COUNT(NONEMPTY(EXISTING [Location].[Coordinates].[Coordinates].MEMBERS,
                   {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

SELECT
    {[Measures].[Latitude Centroid], [Measures].[Record Count],
     [Measures].[Sampled Sites]} ON COLUMNS,
    NONEMPTY(
        EXCEPT([Taxon].[Class].[Class].MEMBERS,
               {[Taxon].[Class].&[__UNKNOWN__]})
        * [Date].[Decade].[Decade].MEMBERS,
        {[Measures].[Record Count]}) ON ROWS
FROM [Marine Range Shift]
WHERE ([Taxon].[Kingdom].&[Animalia]);

-- Q10. Biến động số loài từ 15 độ Bắc theo thập kỷ
-- Decade level must contain every decade and be sorted by numeric key.
-- First decade and comparisons with no observations return NULL.
-- Latitude-band keys 15 and 20 cover >=15N in the study area.

WITH
SET [Known Species] AS
    EXCEPT([Taxon].[Species].[Species].MEMBERS,
           {[Taxon].[Species].&[__UNKNOWN__]})

MEMBER [Measures].[Recorded Species] AS
    COUNT(NONEMPTY(EXISTING [Known Species], {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Previous Decade Species] AS
    IIF([Date].[Decade].CURRENTMEMBER IS [Date].[Decade].&[1980],
        NULL,
        IIF(([Measures].[Record Count], [Date].[Decade].CURRENTMEMBER.LAG(1)) > 0,
            ([Measures].[Recorded Species], [Date].[Decade].CURRENTMEMBER.LAG(1)), NULL)),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Species Change] AS
    IIF([Measures].[Record Count] > 0 AND
        NOT ISEMPTY([Measures].[Previous Decade Species]),
        [Measures].[Recorded Species] - [Measures].[Previous Decade Species], NULL),
    FORMAT_STRING = "+#,##0;-#,##0;0"

MEMBER [Measures].[Species Change Pct] AS
    IIF([Measures].[Record Count] > 0 AND
        [Measures].[Previous Decade Species] > 0 AND
        NOT ISEMPTY([Measures].[Species Change]),
        [Measures].[Species Change] / [Measures].[Previous Decade Species], NULL),
    FORMAT_STRING = "0.00%"

SELECT
    {[Measures].[Recorded Species], [Measures].[Previous Decade Species],
     [Measures].[Species Change], [Measures].[Species Change Pct],
     [Measures].[Record Count]} ON COLUMNS,
    [Date].[Decade].[Decade].MEMBERS ON ROWS
FROM [Marine Range Shift]
WHERE ({[Location].[Latitude Band].&[15], [Location].[Latitude Band].&[20]});


-- Q11. Loài mới được ghi nhận và loài tích lũy từ 20 độ Bắc
-- First means first within the dataset/northern slice, not first arrival.
-- Count the union of species across years; do not SUM annual richness.
-- Do not apply current-year EXISTING to the cumulative species enumeration.
-- Year members start at 1980, ordered by numeric key, including years without observations.

WITH
SET [Known Species] AS
    EXCEPT([Taxon].[Species].[Species].MEMBERS,
           {[Taxon].[Species].&[__UNKNOWN__]})

MEMBER [Measures].[Cumulative Northern Species] AS
    COUNT(NONEMPTY(
        [Known Species],
        {[Date].[Year].&[1980]:[Date].[Year].CURRENTMEMBER}
        * {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[New Northern Species] AS
    [Measures].[Cumulative Northern Species]
    - IIF([Date].[Year].CURRENTMEMBER IS [Date].[Year].&[1980],
          0,
          ([Measures].[Cumulative Northern Species], [Date].[Year].CURRENTMEMBER.LAG(1))),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Sampled Sites] AS
    COUNT(NONEMPTY(EXISTING [Location].[Coordinates].[Coordinates].MEMBERS,
                   {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

SELECT
    {[Measures].[New Northern Species], [Measures].[Cumulative Northern Species],
     [Measures].[Record Count], [Measures].[Sampled Sites]} ON COLUMNS,
    [Date].[Year].[Year].MEMBERS ON ROWS
FROM [Marine Range Shift]
WHERE ([Location].[Latitude Band].&[20]);


-- Q12. Mười lăm loài có khoảng SST ghi nhận rộng nhất
-- Native SST measures use Min/Max aggregation, with NullProcessing=Preserve.
-- This is observed SST span, not physiological tolerance or adaptability.

WITH
SET [Known Species] AS
    EXCEPT([Taxon].[Species].[Species].MEMBERS,
           {[Taxon].[Species].&[__UNKNOWN__]})

MEMBER [Measures].[SST Span] AS
    IIF(NOT ISEMPTY([Measures].[SST Minimum]) AND
        NOT ISEMPTY([Measures].[SST Maximum]),
        [Measures].[SST Maximum] - [Measures].[SST Minimum], NULL),
    FORMAT_STRING = "0.00"

MEMBER [Measures].[Sampled Sites] AS
    COUNT(NONEMPTY(EXISTING [Location].[Coordinates].[Coordinates].MEMBERS,
                   {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

SELECT
    {[Measures].[SST Minimum], [Measures].[SST Maximum], [Measures].[SST Span],
     [Measures].[Record Count], [Measures].[Sampled Sites]} ON COLUMNS,
    TOPCOUNT(
        FILTER(NONEMPTY([Known Species], {[Measures].[Record Count]}),
               NOT ISEMPTY([Measures].[SST Minimum]) AND
               NOT ISEMPTY([Measures].[SST Maximum])),
        15, [Measures].[SST Span]) ON ROWS
FROM [Marine Range Shift];


-- Q13. Tỷ lệ bản ghi Animalia tầng sâu trong số bản ghi biết độ sâu
-- Numerator includes bands 3+4 (>=200m); denominator excludes Unknown.
-- No known depths means NULL percentage, not 0 percent.

WITH
MEMBER [Measures].[Depth Known Records] AS
    COALESCEEMPTY(SUM(
        {[Depth Band].[Band].&[1], [Depth Band].[Band].&[2],
         [Depth Band].[Band].&[3], [Depth Band].[Band].&[4]},
        [Measures].[Record Count]), 0),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Deep Records] AS
    COALESCEEMPTY(SUM({[Depth Band].[Band].&[3], [Depth Band].[Band].&[4]},
                     [Measures].[Record Count]), 0),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Deep Record Pct] AS
    IIF([Measures].[Depth Known Records] > 0,
        COALESCEEMPTY([Measures].[Deep Records], 0) / [Measures].[Depth Known Records], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Depth Known Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY([Measures].[Depth Known Records], 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

SELECT
    {[Measures].[Record Count], [Measures].[Depth Known Records],
     [Measures].[Deep Records], [Measures].[Deep Record Pct],
     [Measures].[Depth Known Pct]} ON COLUMNS,
    NONEMPTY([Date].[Decade].[Decade].MEMBERS
             * [Location].[Latitude Band].[Latitude Band].MEMBERS,
             {[Measures].[Record Count]}) ON ROWS
FROM [Marine Range Shift]
WHERE ([Taxon].[Kingdom].&[Animalia]);


-- Q14. Kiểm toán chất lượng đầu vào đã làm sạch theo quốc gia
-- On-land is a flag, not a confirmed error. Unknown status is not absence.
-- Calculated percentages are not extra physical Fact columns.

WITH
MEMBER [Measures].[On Land Records] AS
    COALESCEEMPTY(([Measures].[Record Count], [Quality].[On Land].&[1]), 0),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Present Records] AS
    COALESCEEMPTY(([Measures].[Record Count], [Quality].[Occurrence Status].&[present]), 0),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[On Land Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY([Measures].[On Land Records], 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Present Status Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY([Measures].[Present Records], 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

SELECT
    {[Measures].[Record Count], [Measures].[On Land Records], [Measures].[On Land Pct],
     [Measures].[Present Records], [Measures].[Present Status Pct]} ON COLUMNS,
    NONEMPTY([Location].[Country].[Country].MEMBERS,
             {[Measures].[Record Count]}) ON ROWS
FROM [Marine Range Shift];


-- Q15. Ghi nhận và tổng individualCount theo quốc gia, vĩ độ, thập kỷ
-- Country share denominator includes Unknown in the same latitude band/decade.
-- Individual Count Records = SUM(1 when individualCount is non-NULL else 0).
-- No count values means NULL total, not a fabricated zero.
-- Reported count sum is not regional population size or standardized abundance.

WITH
SET [Known Species] AS
    EXCEPT([Taxon].[Species].[Species].MEMBERS,
           {[Taxon].[Species].&[__UNKNOWN__]})

MEMBER [Measures].[Recorded Species] AS
    COUNT(NONEMPTY(EXISTING [Known Species], {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Country Record Share] AS
    IIF(([Measures].[Record Count], [Location].[Country].DEFAULTMEMBER) > 0,
        COALESCEEMPTY([Measures].[Record Count], 0) / ([Measures].[Record Count], [Location].[Country].DEFAULTMEMBER), NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Individual Count Coverage] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY([Measures].[Individual Count Records], 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Reported Count Total] AS
    IIF([Measures].[Individual Count Records] > 0,
        [Measures].[Individual Count], NULL),
    FORMAT_STRING = "#,##0.####"

SELECT
    {[Measures].[Record Count], [Measures].[Recorded Species],
     [Measures].[Country Record Share], [Measures].[Reported Count Total],
     [Measures].[Individual Count Records], [Measures].[Individual Count Coverage]} ON COLUMNS,
    NONEMPTY([Location].[Country].[Country].MEMBERS
             * [Location].[Latitude Band].[Latitude Band].MEMBERS
             * [Date].[Decade].[Decade].MEMBERS,
             {[Measures].[Record Count]}) ON ROWS
FROM [Marine Range Shift];
