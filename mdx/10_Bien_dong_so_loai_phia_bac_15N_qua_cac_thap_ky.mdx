-- Adapted to deployed cube MARINE SHIFT RANGE DW / database MARINE_SHIFT_RANGE_SSAS.
-- Q10. Biến động số loài từ 15 độ Bắc theo thập kỷ
-- Decade keys are 1980s..2020s, sorted chronologically in this dataset.
-- First decade and comparisons with no observations return NULL.
-- Latitude-band string keys 15-20°N and 20-23°N cover >=15N in this study area.

WITH
MEMBER [Measures].[Record Count] AS
    [Measures].[Occurrence Count], FORMAT_STRING = "#,##0"

SET [Known Species] AS
    FILTER([Dim Taxon].[Species].[Species].MEMBERS,
           LEN(TRIM([Dim Taxon].[Species].CURRENTMEMBER.PROPERTIES("KEY0"))) > 0)

MEMBER [Measures].[Recorded Species] AS
    COUNT(NONEMPTY(EXISTING [Known Species], {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Previous Decade Species] AS
    IIF([Dim Date].[Decade].CURRENTMEMBER IS [Dim Date].[Decade].&[1980s],
        NULL,
        IIF(([Measures].[Record Count], [Dim Date].[Decade].CURRENTMEMBER.LAG(1)) > 0,
            ([Measures].[Recorded Species], [Dim Date].[Decade].CURRENTMEMBER.LAG(1)), NULL)),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Species Change] AS
    IIF([Measures].[Record Count] > 0 AND
        NOT ISEMPTY([Measures].[Previous Decade Species]),
        [Measures].[Recorded Species] - [Measures].[Previous Decade Species], NULL),
    FORMAT_STRING = "+#,##0;-#,##0;0"

MEMBER [Measures].[Species Change Pct] AS
    IIF([Measures].[Record Count] > 0 AND
        [Measures].[Previous Decade Species] > 0 AND
        NOT ISEMPTY([Measures].[Species Change]),
        [Measures].[Species Change] / [Measures].[Previous Decade Species], NULL),
    FORMAT_STRING = "0.00%"

SELECT
    {[Measures].[Recorded Species], [Measures].[Previous Decade Species],
     [Measures].[Species Change], [Measures].[Species Change Pct],
     [Measures].[Record Count]} ON COLUMNS,
    [Dim Date].[Decade].[Decade].MEMBERS ON ROWS
FROM [MARINE SHIFT RANGE DW]
WHERE ({[Dim Location].[Latitude Band].&[15-20°N], [Dim Location].[Latitude Band].&[20-23°N]});
