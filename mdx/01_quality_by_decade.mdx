-- Q01. Độ đầy đủ và cờ chất lượng theo thập kỷ
-- Taxonomy completeness is measured after Dim_Taxon standardization.
-- Flag Consistency means flag_on_land agrees with the ON_LAND token.
-- This checks warehouse data, not the original unfiltered OBIS export.

WITH
SET [Known Species] AS
    EXCEPT([Taxon].[Species].[Species].MEMBERS,
           {[Taxon].[Species].&[__UNKNOWN__]})

MEMBER [Measures].[Recorded Species] AS
    COUNT(NONEMPTY(EXISTING [Known Species], {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Sampled Sites] AS
    COUNT(NONEMPTY(EXISTING [Location].[Coordinates].[Coordinates].MEMBERS,
                   {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Depth Known Records] AS
    COALESCEEMPTY(SUM(
        {[Depth Band].[Band].&[1], [Depth Band].[Band].&[2],
         [Depth Band].[Band].&[3], [Depth Band].[Band].&[4]},
        [Measures].[Record Count]), 0),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Species Identified Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY(SUM([Known Species], [Measures].[Record Count]), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Complete Taxonomy Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY(([Measures].[Record Count], [Taxon].[Taxonomy Complete].&[1]), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Country Known Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY([Measures].[Record Count] - COALESCEEMPTY(([Measures].[Record Count], [Location].[Country].&[__UNKNOWN__]), 0), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Depth Known Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY([Measures].[Depth Known Records], 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[SST Known Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY([Measures].[Record Count] - COALESCEEMPTY(([Measures].[Record Count], [Location].[SST Group].&[-1]), 0), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[SSS Known Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY([Measures].[Record Count] - COALESCEEMPTY(([Measures].[Record Count], [Location].[SSS Group].&[-1]), 0), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Flagged Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY([Measures].[Record Count] - COALESCEEMPTY(([Measures].[Record Count], [Quality].[Flags].&[[]]]), 0), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[On Land Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY(([Measures].[Record Count], [Quality].[On Land].&[1]), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Flag Consistency Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY(([Measures].[Record Count], [Quality].[Flag Consistency].&[1]), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Depth Issue Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY(([Measures].[Record Count], [Quality].[Depth Issue].&[1]), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Accepted Name Issue Pct] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY(([Measures].[Record Count], [Quality].[Accepted Name Issue].&[1]), 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

SELECT
    {[Measures].[Record Count], [Measures].[Recorded Species], [Measures].[Sampled Sites],
     [Measures].[Species Identified Pct], [Measures].[Complete Taxonomy Pct],
     [Measures].[Country Known Pct], [Measures].[Depth Known Pct],
     [Measures].[SST Known Pct], [Measures].[SSS Known Pct],
     [Measures].[Flagged Pct], [Measures].[On Land Pct],
     [Measures].[Flag Consistency Pct], [Measures].[Depth Issue Pct],
     [Measures].[Accepted Name Issue Pct]} ON COLUMNS,
    NONEMPTY([Date].[Decade].[Decade].MEMBERS,
             {[Measures].[Record Count]}) ON ROWS
FROM [Marine Range Shift];
