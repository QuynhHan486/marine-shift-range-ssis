-- Adapted to deployed cube MARINE SHIFT RANGE DW / database MARINE_SHIFT_RANGE_SSAS.
-- Q15. Ghi nhận và tổng individualCount theo quốc gia, vĩ độ, thập kỷ
-- Country share denominator includes Unknown in the same latitude band/decade.
-- Individual Count Records = SUM(1 when individualCount is non-NULL else 0).
-- No count values means NULL total, not a fabricated zero.
-- Reported count sum is not regional population size or standardized abundance.

WITH
MEMBER [Measures].[Record Count] AS
    [Measures].[Occurrence Count], FORMAT_STRING = "#,##0"

SET [Known Species] AS
    FILTER([Dim Taxon].[Species].[Species].MEMBERS,
           LEN(TRIM([Dim Taxon].[Species].CURRENTMEMBER.PROPERTIES("KEY0"))) > 0)

MEMBER [Measures].[Recorded Species] AS
    COUNT(NONEMPTY(EXISTING [Known Species], {[Measures].[Record Count]})),
    FORMAT_STRING = "#,##0"

MEMBER [Measures].[Country Record Share] AS
    IIF(([Measures].[Record Count], [Dim Location].[Country].DEFAULTMEMBER) > 0,
        COALESCEEMPTY([Measures].[Record Count], 0) / ([Measures].[Record Count], [Dim Location].[Country].DEFAULTMEMBER), NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Individual Count Coverage] AS
    IIF([Measures].[Record Count] > 0,
        COALESCEEMPTY([Measures].[Individual Count Records], 0) / [Measures].[Record Count], NULL),
    FORMAT_STRING = "0.00%"

MEMBER [Measures].[Reported Count Total] AS
    IIF([Measures].[Individual Count Records] > 0,
        [Measures].[Individual Count], NULL),
    FORMAT_STRING = "#,##0.####"

SELECT
    {[Measures].[Record Count], [Measures].[Recorded Species],
     [Measures].[Country Record Share], [Measures].[Reported Count Total],
     [Measures].[Individual Count Records], [Measures].[Individual Count Coverage]} ON COLUMNS,
    NONEMPTY([Dim Location].[Country].[Country].MEMBERS
             * [Dim Location].[Latitude Band].[Latitude Band].MEMBERS
             * [Dim Date].[Decade].[Decade].MEMBERS,
             {[Measures].[Record Count]}) ON ROWS
FROM [MARINE SHIFT RANGE DW];
