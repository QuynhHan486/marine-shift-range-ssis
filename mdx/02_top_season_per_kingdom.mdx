-- Q02. Mùa theo lịch được ghi nhận nhiều nhất của mỗi giới
-- Winter=12/1/2; Spring=3/4/5; Summer=6/7/8; Autumn=9/10/11.
-- TOPCOUNT returns one winning season. A remaining tie is not returned.

WITH
MEMBER [Measures].[Season Record Share] AS
    IIF(([Measures].[Record Count], [Date].[Season].DEFAULTMEMBER) > 0,
        COALESCEEMPTY([Measures].[Record Count], 0) / ([Measures].[Record Count], [Date].[Season].DEFAULTMEMBER), NULL),
    FORMAT_STRING = "0.00%"

SELECT
    {[Measures].[Record Count], [Measures].[Season Record Share]} ON COLUMNS,
    GENERATE(
        EXCEPT([Taxon].[Kingdom].[Kingdom].MEMBERS,
               {[Taxon].[Kingdom].&[__UNKNOWN__]}),
        TOPCOUNT(
            NONEMPTY({[Taxon].[Kingdom].CURRENTMEMBER}
                     * [Date].[Season].[Season].MEMBERS,
                     {[Measures].[Record Count]}),
            1, [Measures].[Record Count])
    ) ON ROWS
FROM [Marine Range Shift];
